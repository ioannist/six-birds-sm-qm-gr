\section*{Step 6 Structural Statement: Bounded \(G_{v2}\) Boundary-Continuum Design}

\begin{definition}[Bounded grammar \(G_{v2}\)]
The grammar \(G_{v2}\) inherits the \(G_{v1}\) source rows \(u_i\) and \(v\).
It permits one additional native source row \(w\), interpreted as a
holographic-RG boundary/continuum flow coordinate under an AdS-like state
class.  It excludes a native triple-package row and excludes a primitive
global compatibility stipulation.
\end{definition}

\begin{definition}[Candidate predicate \(P^{**}\)]
At refinement level \(r\in\{2,4\}\), let the carrier have basis
\[
  u_1,\ldots,u_r,\quad v,\quad w,\quad t,
\]
with \(C=I\).  The native \(G_{v2}\)-lens is
\[
  L_{v2}=\{u_1,\ldots,u_r,v,w\}.
\]
The inherited/generated bridge rows are
\[
  B_{HL}=r^{-1/2}\sum_i u_i,\qquad
  B_{LA}=v,\qquad
  B_{HA}=w.
\]
The triple-package target row is
\[
  B_T=t.
\]
\end{definition}

\begin{proposition}[Partial bridge closure in \(G_{v2}\)]
For \(r=2,4\), \(G_{v2}\) closes
\[
  B_{HL},\quad B_{LA},\quad B_{HA}
\]
with zero residual.  The triple row satisfies
\[
  \frac{\tr\,\Xi_I(B_T\mid L_{v2})}{\tr K_{B_TB_T}}=1.0.
\]
On the coupled target consisting of the generated sector probes and all four
bridge probes,
\[
  \tr\,\Xi_I=1.0,\qquad
  \frac{\tr\,\Xi_I}{\tr K_{DD}}=\frac{1}{7}=0.14285714285714285
\]
at both refinement levels.
\end{proposition}

\begin{proof}
The rows \(B_{HL}\), \(B_{LA}\), and \(B_{HA}\) lie in the span of the declared
native rows \(u_i\), \(v\), and \(w\), respectively, so their Schur residuals
vanish.  The row \(t\) is orthogonal to the native span and is not generated by
\(G_{v2}\), so it contributes one unit of residual trace.  The coupled target
has seven unit-trace diagnostic probes: area, amplitude, P3, and four bridge
rows.  One bridge row remains, giving normalized residual \(1/7\).
\end{proof}

\end{remark}
